rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function memberRole(orgId) {
      return get(/databases/$(database)/documents/orgs/$(orgId)/members/$(request.auth.uid)).data.role;
    }
    function isMember(orgId) {
      return signedIn() && memberRole(orgId) in ['admin', 'staff'];
    }
    function isAdmin(orgId) {
      return signedIn() && memberRole(orgId) == 'admin';
    }

    match /users/{userId}/meta/{docId} {
      allow read, write: if signedIn() && request.auth.uid == userId;
    }
    match /users/{userId}/workspace/{docId} {
      allow read, write: if signedIn() && request.auth.uid == userId;
    }

    match /orgs/{orgId} {
      allow read: if isMember(orgId);
      allow create: if signedIn();
      allow update, delete: if isAdmin(orgId);
    }
    match /orgs/{orgId}/members/{memberId} {
      allow read: if isMember(orgId);
      allow write: if isAdmin(orgId) || (signedIn() && request.auth.uid == memberId);
    }
    match /orgs/{orgId}/workspace/{docId} {
      allow read: if isMember(orgId);
      allow write: if isAdmin(orgId);
    }
    // REQUIRED for large inventory sync (GIMA catalog etc.)
    match /orgs/{orgId}/productChunks/{chunkId} {
      allow read: if isMember(orgId);
      allow write: if isAdmin(orgId);
    }
    match /orgs/{orgId}/pending/{pendingId} {
      allow read: if isMember(orgId);
      allow create: if isMember(orgId);
      allow update, delete: if isAdmin(orgId);
    }
  }
}
